#include <WiFi.h>
#include <Servo.h>
#include "DHT.h"

//configuracion para wifi
const char* ssid = "galaxy";
const char* contrasena = "12345678z";
WiFiServer server(8080);         
WiFiClient client;

Servo servo_0; //declaracion d elos objetos para crontrolar los servos
Servo servo_1; 
Servo servo_2; 
Servo servo_3; 
Servo servo_4; 
Servo servo_5; 

int Led_Rojo = 9; //pines del sensor de color
int Led_Verde = 8;
int Led_Azul = 10;

#define sensor_pin 12   //pin del sensor de proximidad
#define rele_pin 13 

#define DHTPIN 11       //pin del sensor de temperatura
#define DHTTYPE DHT11  

//variables globales
bool modoMulticolor = true;
bool sensorHabilitado = false;
bool releEncendido = false;

DHT dht(DHTPIN, DHTTYPE); 


void setup() {
  Serial.begin(9600); //inicia la comunicacion
  //dht.begin();

  //para conectar a wifi
  WiFi.begin(ssid, contrasena);
  Serial.print("Conectando a la red");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println();
  Serial.print("Conectado a wifi, IP: ");
  Serial.println(WiFi.localIP());
  server.begin(); //se inicia el servidor TCP

  servo_0.attach(2); //El servo 0 al pin 2
  servo_1.attach(3); //El servo 1 al pin 3
  servo_2.attach(4); //El servo 2 al pin 4
  servo_3.attach(5); //El servo 3 al pin 5
  servo_4.attach(6); //El servo 4 al pin 6
  servo_5.attach(7); //El servo 5 al pin 7

  //pines del sensor de color
  pinMode(Led_Rojo, OUTPUT); 
  pinMode(Led_Verde, OUTPUT); 
  pinMode(Led_Azul, OUTPUT); 

  //pines del sensor y el relevador
  pinMode(sensor_pin, INPUT);
  pinMode(rele_pin, OUTPUT);

  digitalWrite(rele_pin, LOW);
}

void loop() {
  if (!client || !client.connected()){
        client = server.available();
    }
  
  while (Serial.available()) {
    char comando = Serial.read();
    int valor = 0;
    if (comando >= 'A' && comando <= 'F') {
      while (Serial.available() == 0) { delay(1); }
        valor = Serial.read();
    }
    procesarComando(comando, valor);
  }
  while (client && client.connected() && client.available()) {
    char comando = client.read();
    int valor = 0;
    if (comando >= 'A' && comando <= 'F') {
      while (!client.available()) { delay(1); }
        valor = client.read();
    }
    procesarComando(comando, valor);
  }

  if (modoMulticolor) {
    multicolor();
  }

  if (sensorHabilitado && releEncendido) {
    int estado = digitalRead(sensor_pin);

    if (estado == LOW) {  // si detecta objeto
      digitalWrite(rele_pin, LOW); // apaga relevador solo mientras detecta
    } else {
      digitalWrite(rele_pin, HIGH); // vuelve a encenderse cuando no hay objeto
    }
  }
}

void procesarComando(char comando, int valor) {
  switch (comando) {
    case 'A': { 
      servo_0.write(constrain(valor, 0, 90));             //aqui se mueve el servomotor
      break;
    }
    case 'B': { 
      servo_1.write(constrain(valor, 0, 180));             //aqui se mueve el servomotor
      break;
    }
    case 'C': { 
      servo_2.write(constrain(valor, 0, 180));              //aqui se mueve el servomotor
      break;
    }
    case 'D': { 
      servo_3.write(constrain(valor, 0, 180));              //aqui se mueve el servomotor
      break;
    }
    case 'E': { 
      servo_4.write(constrain(valor, 0, 180));             //aqui se mueve el servomotor 
      break;
    }
    case 'F': { 
      servo_5.write(constrain(valor, 0, 180));              //aqui se mueve el servomotor
      break;
    }
    case 'G': {
      amarillo();
      modoMulticolor = false;
      break;
    }
    case 'H': {
      rojo();
      modoMulticolor = false;
      break;
    }
    case 'V': {
      verde();
      modoMulticolor = false;
      break;
    }
    case 'J': {
      azul();
      modoMulticolor = false;
      break;
    }
    case 'K': {
      multicolor();
      modoMulticolor = true;
      break;
    }
    case 'N': {
      naranja();
      modoMulticolor = false;
      break;
    }
    case 'M': {
      morado();
      modoMulticolor = false;
      break;
    }
    case 'S': {
      sensorHabilitado = true;
      releEncendido = true;
      digitalWrite(rele_pin, HIGH);
      break;
    }
    case 'W': {
      sensorHabilitado = false;
      releEncendido = false;
      digitalWrite(rele_pin, LOW);
      break;
    }
  }
}
